11_210xx_mass_assign_hpp_headers.conf

File rule ModSecurity (bấm nút Copy ở góc phải để dán thẳng lên WAF UI):

# Config Type trên WAF UI: MODSEC (Service: Global)
# ════════ 210xx — MASS ASSIGNMENT / HPP / AUTH BYPASS HEADERS ═════════
# Mass Assignment: chặn field đặc quyền ở route đăng ký/profile.
# (?:^|[.\[])…(?:\]|\.array_\d+|$) để bắt cả "role", "json.role", "json.role.array_0", "json.profile.role" và PHP form "user[role]".
SecRule REQUEST_FILENAME "@rx (?i)^/(?:api/)?(?:register|signup|profile|users?/update|account)/?$" \
    "id:21001,phase:2,deny,status:403,log,t:none,t:normalizePath,chain,msg:'Privileged field mass assignment',tag:'mass-assignment',logdata:'%{MATCHED_VAR_NAME}'"
    SecRule ARGS_NAMES "@rx (?i)(?:^|[.\[])(?:role|is_admin|isadmin|admin|balance|credit|credits|permissions?|user_id|owner_id|verified|email_verified)(?:\]|\.array_\d+|$)" "t:none"

# HPP dạng query/form: ?id=1&id=2.
SecRule ARGS:id "@rx ." \
    "id:21002,phase:2,deny,status:400,log,t:none,chain,msg:'Duplicate id parameter (HPP)',tag:'hpp'"
    SecRule &ARGS:id "!@eq 1"

# HPP dạng JSON array: {"id":[1,2]} → json.id.array_0, json.id.array_1.
# Chỉ bật nếu app không có nhu cầu nhận mảng id hợp lệ.
SecRule ARGS_NAMES "@rx ^json\.id\.array_(?:[1-9][0-9]*)$" \
    "id:21004,phase:2,deny,status:400,log,t:none,msg:'Duplicate id in JSON array (HPP)',tag:'hpp',logdata:'%{MATCHED_VAR_NAME}'"

# Header giả mạo đường dẫn nội bộ. Để pass,log vì proxy BTC có thể tự gắn
# X-Forwarded-Host; chỉ chuyển deny với X-Original-URL|X-Rewrite-URL nếu thấy bị lạm dụng.
SecRule REQUEST_HEADERS_NAMES "@rx (?i)^(?:x-original-url|x-rewrite-url|x-custom-ip-authorization|x-forwarded-host)$" \
    "id:21003,phase:1,pass,log,t:none,msg:'Suspicious override header (detect)',tag:'auth-bypass',logdata:'%{MATCHED_VAR_NAME}'"
