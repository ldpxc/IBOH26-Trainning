00_disable_fp_crs.conf

File rule ModSecurity (bấm nút Copy ở góc phải để dán thẳng lên WAF UI):

# ==============================================================================
# 00_disable_fp_crs.conf — Tắt các rule OWASP CRS (9xxxxx) hay chặn nhầm Bot SVC
# Config Type trên WAF UI: MODSEC (Service: Global) — BẮT BUỘC nạp SAU CRS!
# ==============================================================================
# Lưu ý:
#   - Nhóm 1 & 2 đang BẬT SẴN: an toàn, giảm tối đa nguy cơ rớt điểm SVC.
#   - Nhóm 3 đang ĐỂ COMMENT (#): chỉ bỏ dấu # nếu thấy đúng ID đó chặn nhầm
#     bot trên Traffic log (lọc status:403).
# ==============================================================================
# Configs → New → Service: Global → Config Type: MODSEC (bắt buộc MODSEC để chạy SAU OWASP CRS thì SecRuleRemoveById mới có tác dụng) → đặt tên 00-disable-fp-crs → dán nội dung file → Save.

# ─── NHÓM 1: BẬT NGAY (Các rule CRS hay báo nhầm nhất với Bot SVC phòng thi) ───

# 920350: Host header là địa chỉ IP (vd 127.0.0.1, 10.13.x.x) — hay cộng +3 điểm
# cho 100% traffic và làm rác Traffic log.
SecRuleRemoveById 920350

# 920300 / 920310 / 920311: Thiếu hoặc rỗng header Accept (script bot Python/Go hay không gửi).
SecRuleRemoveById 920300
SecRuleRemoveById 920310
SecRuleRemoveById 920311

# 920320 / 920330: Thiếu hoặc rỗng header User-Agent (một số healthcheck tối giản).
SecRuleRemoveById 920320
SecRuleRemoveById 920330


# ─── NHÓM 2: GỠ COOKIE SESSION & JWT KHỎI RULE CRS HAY DÍNH CHUỖI NGẪU NHIÊN ───
# Chuỗi session/JWT (Base64URL có ký tự --, _, chữ số) rất hay bị CRS 942xxx/941xxx
# nhận nhầm thành SQL comment hoặc XSS.

SecRuleUpdateTargetById 942100 "!REQUEST_COOKIES:PHPSESSID"
SecRuleUpdateTargetById 942440 "!REQUEST_COOKIES:PHPSESSID"
SecRuleUpdateTargetById 942450 "!REQUEST_COOKIES:PHPSESSID"
SecRuleUpdateTargetById 941100 "!REQUEST_COOKIES:PHPSESSID"

SecRuleUpdateTargetById 942100 "!REQUEST_COOKIES:session"
SecRuleUpdateTargetById 942440 "!REQUEST_COOKIES:session"
SecRuleUpdateTargetById 942450 "!REQUEST_COOKIES:session"
SecRuleUpdateTargetById 941100 "!REQUEST_COOKIES:session"

SecRuleUpdateTargetById 942100 "!REQUEST_HEADERS:Authorization"
SecRuleUpdateTargetById 942440 "!REQUEST_HEADERS:Authorization"
SecRuleUpdateTargetById 942450 "!REQUEST_HEADERS:Authorization"


# ─── NHÓM 3: DỰ PHÒNG (Bỏ dấu # CHỈ KHI thấy đúng ID đó chặn nhầm bot SVC) ────

# 920420: Content-Type không nằm trong danh sách mặc định (vd text/plain, application/graphql)
# SecRuleRemoveById 920420

# 920170 / 920171: Request GET/HEAD có gửi kèm Body
# SecRuleRemoveById 920170
# SecRuleRemoveById 920171

# 920230: Phát hiện URL encode 2 lớp (hay dính nếu app truyền redirect URL lồng URL)
# SecRuleRemoveById 920230

# 920270 / 920271: Ký tự ngoài dải ASCII (hay dính nếu bot gửi tiếng Việt/UTF-8 đặc biệt)
# SecRuleRemoveById 920270
# SecRuleRemoveById 920271

# 913100 / 913101 / 913102: Chặn User-Agent dạng tool/script (bỏ # nếu UA bot bị nhận nhầm)
# SecRuleRemoveById 913100
# SecRuleRemoveById 913101
# SecRuleRemoveById 913102

# 942430 / 942431: Đếm số ký tự đặc biệt SQL (hay chặn nhầm ô nhập text dài/JSON phức tạp)
# SecRuleRemoveById 942430
# SecRuleRemoveById 942431
