# ==============================================================================
# F-HACKER 2026 — ModSecurity v3 Defense Ruleset (Tested with waf_rule_tester.py)
# Config Type: MODSEC (nạp sau OWASP CRS) | Dải Rule ID: 27001–27008
# ==============================================================================

# [27001] SQL Injection (SQLi)
# Chặn: boolean tautology (' OR 1=1, 1 OR 1=1), UNION SELECT, comment SQL (--/#/*),
#       stacked query (; DROP/DELETE/INSERT/UPDATE), time-based (SLEEP/BENCHMARK/WAITFOR),
#       subquery (SELECT ... FROM) — không chặn nhầm text thường ("SELECT * FROM", "union all", "insert coin")
SecRule ARGS|ARGS_NAMES|REQUEST_BODY "@rx (?:(?:\x27|\x22)\s*(?:or|and)\s+(?:\x27[^\x27]{0,16}\x27|\d+)\s*=\s*(?:\x27[^\x27]{0,16}\x27?|\d+)|(?:\b\d+\b|\x27|\x22)\s+(?:or|and)\s+\d+\s*=\s*\d+|\bunion\b[\s/+*]{1,16}(?:all[\s/+*]{1,8})?\bselect\b|\b(?:sleep|benchmark|pg_sleep|extractvalue|updatexml|load_file)\s*\(|\bwaitfor\s+delay\b|;\s*(?:drop|delete|alter|truncate|create|insert|update|exec)\b|\(\s*select\b.{0,80}\bfrom\b|\x27\s*(?:--|#|/\*)|\b(?:or|and)\s+\x27[^\x27]{0,16}\x27\s*=\s*\x27[^\x27]{0,16}\x27?|(?:\x27|\x22)\s*(?:or|and)\s+(?:true|false|1)\b|/\*!\d{0,5}(?:union|select|and|or)|\b(?:information_schema|sqlite_master|pg_catalog)\b)" \
    "id:27001,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,t:urlDecodeUni,msg:'SQLi attack detected',tag:'sqli',logdata:'%{MATCHED_VAR_NAME}=%{MATCHED_VAR}'"

# [27002] OS Command Injection (CMDi)
# Chặn: ; cmd, | cmd, `cmd`, $(cmd), && cmd, || cmd, & <shell_cmd> (không chặn nhầm query param &password=...)
SecRule ARGS|REQUEST_BODY "@rx (?:[;|]\s*(?:/?(?:usr/)?(?:s?bin/)?[a-z]{2,15}\b)|`[^`]{1,120}`|\$\([^)]{1,120}\)|\$\{IFS\}|\$IFS\b|(?:&&|\|\|)\s*[a-z/]|(?:^|\s)&\s*(?:ping|id|whoami|uname|cat|ls|sh|bash|curl|wget|nc|ncat|python3?|perl|php|sleep|echo|dir|type|net)\b)" \
    "id:27002,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,t:urlDecodeUni,msg:'CMDi attack detected',tag:'cmdi',logdata:'%{MATCHED_VAR_NAME}=%{MATCHED_VAR}'"

# [27003] LFI / Path Traversal
# Chặn: ../, ..\, ....//, %2e%2e%2f, %252e%252e, php://, phar://, file://, /etc/passwd, /etc/shadow, /proc/self
SecRule ARGS|REQUEST_URI "@rx (?:\.\.(?:[/\\]|%2f|%5c)|%2e%2e(?:%2f|%5c|[/\\])|%252e%252e|\.\.\.\.[/\\]|\b(?:php|phar|zip|expect|data|file)://|/(?:etc/(?:passwd|shadow|hosts)|proc/(?:self|\d+)/(?:environ|cmdline))|(?:\.\.[/\\]){1,}|\\windows\\win\.ini)" \
    "id:27003,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,t:urlDecodeUni,msg:'LFI or path traversal detected',tag:'lfi',logdata:'%{MATCHED_VAR_NAME}=%{MATCHED_VAR}'"

# [27004] Cross-Site Scripting (XSS)
# Chặn: thẻ thực thi (<script, <svg, <img, <iframe, <body, <object, <embed, <details, <math, <video, <audio, <input, <marquee),
#       event handler (onXXX=), javascript:/vbscript:/data:text/html
SecRule ARGS|REQUEST_BODY "@rx (?:<\s*/?\s*(?:script|svg|iframe|object|embed|applet|meta|link|style|base|details|math|marquee|body|video|audio|form)\b|<\s*(?:img|input|a|div|button|textarea|select|keygen|isindex)\b[^>]{0,200}\bon[a-z]{2,20}\s*=|(?:[\s/\x22\x27>]|^)on(?:error|load|click|mouseover|focus|blur|toggle|animationstart|pointerover|mouseenter|input|submit|change|keydown|keyup|pageshow|beforetoggle)\s*=|\b(?:javascript|vbscript)\s*:|data\s*:\s*text/html)" \
    "id:27004,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,t:urlDecodeUni,t:htmlEntityDecode,t:jsDecode,msg:'XSS attack detected',tag:'xss',logdata:'%{MATCHED_VAR_NAME}=%{MATCHED_VAR}'"

# [27005] Server-Side Template Injection (SSTI)
# Chặn: {{...}}, {%...%}, ${...}, #{...}, <%...%>, *{...}, và chuỗi truy cập object Python/Java
SecRule ARGS|REQUEST_URI|REQUEST_BODY "@rx (?:\{\{[^}]{1,250}\}\}|\{%[^%]{1,250}%\}|\$\{[^}]{1,200}\}|#\{[^}]{1,200}\}|<%[^%]{1,200}%>|\*\{[^}]{1,200}\}|__(?:class|mro|bases|subclasses|globals|builtins|init|import)__)" \
    "id:27005,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,t:urlDecodeUni,msg:'SSTI template injection detected',tag:'ssti',logdata:'%{MATCHED_VAR_NAME}=%{MATCHED_VAR}'"

# [27006] Server-Side Request Forgery (SSRF)
# Chặn: scheme nguy hiểm (gopher/dict/file/ftp/ldap) và URL trỏ loopback/private IP/cloud metadata
SecRule ARGS|REQUEST_BODY "@rx (?:\b(?:gopher|dict|file|ftp|ftps|sftp|tftp|ldap|ldaps|jar)://|https?://(?:[^/\s@]*@)?(?:\[?(?:localhost|0\.0\.0\.0|127(?:\.\d{1,3}){1,3}|10(?:\.\d{1,3}){3}|172\.(?:1[6-9]|2\d|3[01])(?:\.\d{1,3}){2}|192\.168(?:\.\d{1,3}){2}|169\.254\.169\.254|::1|::ffff:127\.0\.0\.1|metadata\.google\.internal)\]?)(?:[:/?#]|$))" \
    "id:27006,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,t:urlDecodeUni,msg:'SSRF internal or dangerous scheme detected',tag:'ssrf',logdata:'%{MATCHED_VAR_NAME}=%{MATCHED_VAR}'"

# [27007] Flag Leak Protection (Phase 4 — RESPONSE_BODY)
# Chặn rò rỉ chuỗi cờ định dạng FHACKER2026{...}
SecRule RESPONSE_BODY "@rx FHACKER2026\{[A-Za-z0-9_=-]{8,128}\}" \
    "id:27007,phase:4,deny,status:403,log,t:none,msg:'Flag leak detected in response body',tag:'flag-leak'"

# [27008] Recon & Sensitive File Exposure
# Chặn truy cập /.env, /.git/*, /.DS_Store, /.svn, /.htaccess, *.bak, *.sql, *.old, *.swp, backup DB...
SecRule REQUEST_FILENAME "@rx (?:^|/)\.(?:git|svn|hg|env|htaccess|htpasswd|DS_Store|ssh|bash_history|bashrc|profile|npmrc|dockerenv|aws)(?:/|\.|$)|\.(?:bak|backup|old|orig|save|swp|swo|sql|sqlite|sqlite3|db|dump|tar\.gz|tgz|zip|7z|rar|dist|inc|env)$" \
    "id:27008,phase:1,deny,status:403,log,t:none,t:urlDecodeUni,t:normalizePath,msg:'Recon sensitive file probe detected',tag:'recon',logdata:'%{REQUEST_FILENAME}'"
