05_204xx_rce_upload.conf

File rule ModSecurity (bấm nút Copy ở góc phải để dán thẳng lên WAF UI):

# Config Type trên WAF UI: MODSEC (Service: Global)
# ═════════════ 204xx — RCE / COMMAND INJECTION / UPLOAD ════════════════
# Allowlist host/IP cho endpoint ping. Sửa route cho đúng app.
SecRule REQUEST_FILENAME "@rx (?i)^/api/ping/?$" \
    "id:20400,phase:2,deny,status:403,log,t:none,t:normalizePath,chain,msg:'Invalid ping host',tag:'rce',logdata:'host=%{MATCHED_VAR}'"
    SecRule ARGS:host|ARGS:ip|ARGS:target|ARGS:domain|ARGS:json.host|ARGS:json.ip|ARGS:json.target|ARGS:json.domain "!@rx ^(?:[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}[A-Za-z0-9])?\.)*[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}[A-Za-z0-9])?$|^(?:\d{1,3}\.){3}\d{1,3}$" "t:none"

# Shell metacharacter trong tham số nghi chạy lệnh (form + JSON).
SecRule ARGS:cmd|ARGS:command|ARGS:exec|ARGS:ping|ARGS:host|ARGS:ip|ARGS:domain|ARGS:target|ARGS:filename|ARGS:json.cmd|ARGS:json.command|ARGS:json.exec|ARGS:json.ping|ARGS:json.host|ARGS:json.ip|ARGS:json.domain|ARGS:json.target|ARGS:json.filename \
    "@rx [;&|`\r\n<>]|\$\(|\$\{|\|\||&&|\x00" \
    "id:20401,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,t:urlDecodeUni,msg:'Shell metacharacter injection',tag:'rce',logdata:'%{MATCHED_VAR_NAME}=%{MATCHED_VAR}'"

# Hàm thực thi mã phổ biến. Có thể false positive trên ô "code"; giới hạn route nếu cần.
SecRule ARGS:cmd|ARGS:code|ARGS:exec|ARGS:eval|ARGS:template|ARGS:expr|ARGS:formula|ARGS:content|ARGS:json.cmd|ARGS:json.code|ARGS:json.exec|ARGS:json.eval|ARGS:json.template|ARGS:json.expr|ARGS:json.formula|ARGS:json.content \
    "@rx (?i)(?:\b(?:system|exec|shell_exec|passthru|popen|proc_open|pcntl_exec)\s*\(|\b(?:eval|assert|create_function)\s*\(|\b(?:include|require)(?:_once)?\s*\(|\b(?:base64_decode|gzinflate|str_rot13|convert_uudecode)\s*\(|java\.lang\.(?:Runtime|ProcessBuilder)|getRuntime\s*\(\s*\)\s*\.\s*exec|ProcessBuilder\s*\(|jndi\s*:\s*(?:ldap|rmi|dns|iiop)|InitialContext|lookup\s*\(\s*[\x22\x27](?:ldap|rmi)://)" \
    "id:20402,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,t:urlDecodeUni,t:lowercase,msg:'Code execution pattern',tag:'rce',logdata:'%{MATCHED_VAR_NAME}=%{MATCHED_VAR}'"

# Backtick command substitution ở BẤT KỲ tham số nào (kể cả ?x=`id`).
# Nếu app có ô markdown dùng `code`, hãy đổi ARGS thành danh sách tham số hẹp.
SecRule ARGS "@rx `[^`]{1,200}`" \
    "id:20403,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,t:urlDecodeUni,msg:'Backtick command substitution',tag:'rce',logdata:'%{MATCHED_VAR_NAME}=%{MATCHED_VAR}'"

# Command substitution $(...) ở bất kỳ tham số nào.
SecRule ARGS "@rx \$\([^)]{1,200}\)" \
    "id:20404,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,t:urlDecodeUni,msg:'Command substitution $()',tag:'rce',logdata:'%{MATCHED_VAR_NAME}=%{MATCHED_VAR}'"

# Upload: allowlist đuôi file. FILES = tên file gốc (KHÔNG phải FILES_NAMES).
SecRule REQUEST_FILENAME "@rx (?i)^/(?:api/)?upload(?:/|$)" \
    "id:20410,phase:2,deny,status:403,log,t:none,t:normalizePath,chain,msg:'Disallowed upload extension',tag:'upload',logdata:'%{FILES}'"
    SecRule FILES "!@rx ^[A-Za-z0-9._-]{1,100}\.(?:jpg|jpeg|png|gif|webp|pdf|txt|csv)$" "t:none,t:lowercase"

# Upload: double extension (shell.php.jpg).
SecRule FILES "@rx (?i)\.(?:php\d?|phtml|phar|jsp|jspx|asp|aspx|ashx|asmx|cgi|pl|py|sh|exe|dll|so)\.(?:[A-Za-z0-9]{1,8})$" \
    "id:20411,phase:2,deny,status:403,log,t:none,t:lowercase,msg:'Double extension upload',tag:'upload',logdata:'%{FILES}'"

# Upload: path traversal / null byte / .htaccess / web.config trong tên file.
SecRule FILES "@rx (?i)(?:[/\x5c]|\.\.|%00|\x00|(?:^|/)\.htaccess$|(?:^|/)web\.config$|\.\s*$)" \
    "id:20412,phase:2,deny,status:403,log,t:none,t:urlDecodeUni,msg:'Malicious upload filename',tag:'upload',logdata:'%{FILES}'"

# Webshell content — cách BỀN nhất trên BunkerWeb: soi REQUEST_BODY của multipart.
# REQUEST_BODY chứa cả phần nội dung file (đã kiểm chứng trên image BTC), không cần
# bật SecUploadKeepFiles/SecTmpSaveUploadedFiles. Chỉ áp dụng cho route upload.
SecRule REQUEST_FILENAME "@rx (?i)^/(?:api/)?upload(?:/|$)" \
    "id:20414,phase:2,deny,status:403,log,t:none,t:normalizePath,chain,msg:'Webshell content in upload body',tag:'upload',logdata:'%{REQUEST_URI}'"
    SecRule REQUEST_BODY "@rx (?i)(<\?php|<\?=|<%|<jsp|Runtime\.getRuntime|ProcessBuilder|cmd\.exe|powershell|/bin/(?:ba)?sh|GIF89a<\?php)" "t:none"

# Webshell content trong BẤT KỲ multipart upload (không phụ thuộc route cụ thể).
# Chỉ soi khi Content-Type là multipart/form-data nên không ảnh hưởng JSON/form thường.
SecRule REQUEST_HEADERS:Content-Type "@rx (?i)^multipart/form-data" \
    "id:20415,phase:2,deny,status:403,log,t:none,chain,msg:'Webshell content in multipart body',tag:'upload',logdata:'%{REQUEST_URI}'"
    SecRule REQUEST_BODY "@rx (?i)(<\?php|<\?=|<%|<jsp|Runtime\.getRuntime|ProcessBuilder|cmd\.exe|powershell|/bin/(?:ba)?sh|GIF89a<\?php)" "t:none"

# (Tùy chọn) FILES_TMP_CONTENT chỉ có dữ liệu khi bật SecUploadKeepFiles On +
# SecUploadDir (hoặc SecTmpSaveUploadedFiles On). Trên BunkerWeb, bật
# SecTmpSaveUploadedFiles On có thể làm multipart trả 400 nếu SecTmpDir không
# ghi được — vì vậy rule 20414 ở trên an toàn hơn cho môi trường thi.
SecRule FILES_TMP_CONTENT \
    "@rx (?i)(<\?php|<\?=|<%|<jsp|Runtime\.getRuntime|ProcessBuilder|cmd\.exe|powershell|/bin/(?:ba)?sh|GIF89a<\?php)" \
    "id:20413,phase:2,deny,status:403,log,t:none,msg:'Webshell content (FILES_TMP_CONTENT)',tag:'upload',logdata:'%{FILES}'"
