07_206xx_xxe.conf

File rule ModSecurity (bấm nút Copy ở góc phải để dán thẳng lên WAF UI):

# Config Type trên WAF UI: MODSEC (Service: Global)
# ═══════════════════════════ 206xx — XXE ══════════════════════════════
# Ép XML body sang URLENCODED để REQUEST_BODY giữ nguyên chuỗi <!ENTITY.
# Nếu app thật cần CRS parse XML, phải COMMENT/XÓA rule này (chỉ đổi pass,log không tắt được ctl:).
SecRule REQUEST_HEADERS:Content-Type "@rx (?i)(?:application|text)/(?:xml|soap\+xml)" \
    "id:20602,phase:1,pass,nolog,t:none,ctl:requestBodyProcessor=URLENCODED"

# XXE trong XML body.
SecRule REQUEST_HEADERS:Content-Type "@rx (?i)(?:application|text)/(?:xml|soap\+xml)" \
    "id:20600,phase:2,deny,status:403,log,t:none,chain,msg:'XXE attempt',tag:'xxe',logdata:'%{MATCHED_VAR_NAME}'"
    SecRule REQUEST_BODY "@rx (?i)(<!ENTITY|<!DOCTYPE[^>]*\[|SYSTEM\s+[\x22\x27]|PUBLIC\s+[\x22\x27]|file://|php://|expect://|jar:|gopher://)" "t:none"

# XXE giấu trong JSON string / tham số.
SecRule ARGS|REQUEST_BODY "@rx (?i)(<!ENTITY|SYSTEM\s+[\x22\x27](?:file|php|expect|jar|gopher|https?)://)" \
    "id:20601,phase:2,deny,status:403,log,t:none,msg:'XXE entity in payload',tag:'xxe',logdata:'%{MATCHED_VAR_NAME}'"
